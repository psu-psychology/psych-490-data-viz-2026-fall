days <- factor(
  c("sun", "mon", "tue", "wed", "thu", "fri", "sat"),
  levels = c("sun", "mon", "tue", "wed", "thu", "fri", "sat"),
  ordered = TRUE
)
meals <- factor(
  c("breakfast", "lunch", "dinner", "snack", "dessert"),
  levels = c("breakfast", "lunch", "snack", "dinner", "dessert"),
  ordered = TRUE
)
meal_days <- rep(days, each = length(meals))
weekly_meals <- rep(meals, 7)

my_meals <- data.frame(day_of_wk = meal_days, 
                       meal = weekly_meals)

                 replace = TRUE, prob = c(.6, .4))
my_meals$ate_it <- ate_it

meals_eaten <- my_meals$meal[my_meals$ate_it == TRUE]
meals_eaten_df <- data.frame(meal = meals_eaten)

Components

  • Mappings
    • Counts \(\rightarrow\) height, length or area; color; texture
      • Sort order \(\rightarrow\) relative position

  • Caption & titles

## Pie

```{r}
#| label: fig-nominal-piechart
#| fig-cap: "A pie chart of the random favorite color data."
#| out-width: "70%"
colors_xtab <- data.frame(table(colors_df))
names(colors_xtab) <- c("favorite_color", "count")

colors_xtab |>
  ggplot(aes(x = "", y = count, fill = favorite_color)) +
  geom_bar(stat = "identity", width = 1) +
  coord_polar("y", start = 0) +
  scale_fill_identity()
```

## Ring

```{r}
#| label: fig-nominal-ring-chart
#| fig-cap: "A ring/donut chart of the random favorite color data."
#| out-width: "70%"

# https://r-graph-gallery.com/128-ring-or-donut-plot.html

# Compute percentages
colors_xtab$fraction = colors_xtab$count / sum(colors_xtab$count)

# Compute the cumulative percentages (top of each rectangle)
colors_xtab$ymax = cumsum(colors_xtab$fraction)

# Compute the bottom of each rectangle
colors_xtab$ymin = c(0, head(colors_xtab$ymax, n = -1))

# Make the plot
colors_xtab |>
  ggplot(aes(
    ymax = ymax,
    ymin = ymin,
    xmax = 4,
    xmin = 3,
    fill = favorite_color
  )) +
  geom_rect() +
  coord_polar(theta = "y") + # Try to remove that to understand how the chart is built initially
  xlim(c(2, 4)) + # Try to remove that to see how to make a pie chart
  scale_fill_identity()
```

## Barplot-1

```{r}
#| label: fig-nominal-barplot
#| fig-cap: "A barplot of the random favorite color data."
#| out-width: "70%"
colors_df |>
  ggplot() +
  aes(x = favorite_color, fill = favorite_color) +
  geom_bar() +
  scale_discrete_manual(aesthetics = c("color", "fill"),
                        values = sort(colors))
```

## Barplot-2

```{r}
#| label: fig-nominal-barplot-ordered
#| fig-cap: "A colorful barplot of the random favorite color data sorted by the number of responses."
#| out-width: "70%"
colors_df |>
  dplyr::count(favorite_color) |>
  dplyr::arrange(desc(n)) |>
  dplyr::mutate(favorite_color = factor(favorite_color, levels = favorite_color)) |>
  ggplot() +
  geom_bar(aes(x = favorite_color, y = n, fill = favorite_color), stat = "identity") +
  scale_fill_identity()
```

# Components of data figures

## Components

- Axes
  - Scales (discrete/nominal) vs. continuous
- Mappings
  - Counts $\rightarrow$ height, length or area; color; texture
    - Sort order $\rightarrow$ relative position
  - Values $\rightarrow$ position
- Legends
- Caption & titles
