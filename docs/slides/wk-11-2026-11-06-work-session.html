if (!dir.exists('../include/csv')) {
  message("Creating missing `csv/`.")

if (params$update_data) {
  options(gargle_oauth_email = Sys.getenv("GMAIL_SURVEY"))
  googledrive::drive_auth()

  googledrive::drive_download(
    "life-gotten-expensive-nytimes",
    path = file.path("../include/csv", params$fn),
    type = "csv",
    overwrite = TRUE
  )
  message("Data updated.")
} else {
  message("Using stored data.")

Visualization: Housing

expensive_df |>
  dplyr::filter(category == "housing") |>
  ggplot() +
  aes(x = year, y = price, color = location, 
      fill = location, shape = price_type) |>
  geom_point() + 
  geom_line(aes(x = year, y = price, 
                group = location, color = location)) +
  ylim(0, 1700000) +
  theme_classic() +
  labs(title = "Price changes in housing expenses.", 
       caption = "Source: @Tavernise2026-zb.")
