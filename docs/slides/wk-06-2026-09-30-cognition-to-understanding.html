-- @Franconeri2021-uv

```{r}
#| label: fig-datasaurus-dozen
#| fig-cap: "The DatasauRus dozen [@datasaurus2019-us; @Matejka2017-xi; @datasauRus]"
library(datasauRus)
library(ggplot2)
library(gganimate)

ggplot(datasaurus_dozen, aes(x=x, y=y))+
  geom_point()+
  theme_minimal() +
  transition_states(dataset, 3, 1) + 
  ease_aes('cubic-in-out')
```
