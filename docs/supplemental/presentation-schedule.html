## Background

This page documents how the presentation schedule can be automatically generated from the student survey.

## Download data

```{r}
googledrive::drive_auth(email = "rick.o.gilmore@gmail.com")
```

```{r}
if (!dir.exists('csv')) {
  dir.create('csv')
}

csv_fn <- "csv/psych-490-2026-fall-presentations.csv"

googledrive::drive_download(
  file = 'PSYCH 490.001 Fall 2026 Project Info (Responses)',
  path = csv_fn,
  type = 'csv',
  overwrite = TRUE
  
)
```

```{r}
if (file.exists(csv_fn)) {
  presentations_df <- readr::read_csv(csv_fn, show_col_types = FALSE)
} else {
  message("File not found: ", csv_fn)
  presentations_df <- NULL
}
```

## Clean the data

```{r}
new_names <-
  c(
    "Timestamp",
    "Authors",
    "Title",

# Make new data frame with long and short names for reference
presentation_qs <- tibble::tibble(q_long = names(presentations_df), q_short = new_names)
```

## Select subsets by preferred date

I assigned students by preferred date and the date they completed the survey.
Students who indicated no preference were added to open spots in the schedule in the order that they completed the survey.

```{r}
mon_07_df <- presentations_df |>
  dplyr::filter(stringr::str_detect(Preferred_date, "07"),
                Give_talk == "Yes")

wed_09_df <- presentations_df |>
  dplyr::filter(stringr::str_detect(Preferred_date, "09"),
                Give_talk == "Yes")

fri_11_df <- presentations_df |>
  dplyr::filter(stringr::str_detect(Preferred_date, "11"),
                Give_talk == "Yes")

no_pref <- presentations_df |>
  dplyr::filter(stringr::str_detect(Preferred_date, "No preference"),
                Give_talk == "Yes")

times <- c("03:40", "03:55", "04:10")
```

## Day 1: Mon, Dec 07

```{r}
#| label: tbl-mon-dec-07
#| tbl-cap: "Presenters for Monday, December 7, 2026"
n_mon_07_df <- dim(mon_07_df)[1]
n_no_pref <- dim(no_pref)[1]

n_mon_07_slots_free <- length(times) - n_mon_07_df

if (n_mon_07_slots_free > 0) {
  new_mon_07_df <- rbind(mon_07_df, no_pref[1:n_mon_07_slots_free,])
} else {
  new_mon_07_df <- mon_07_df
}
new_mon_07_df$Time <- times

no_pref$assigned[1:n_mon_07_slots_free] <- TRUE

new_mon_07_df |>
  dplyr::filter(!is.na(Authors)) |>
  dplyr::arrange(Time) |>
  dplyr::select(Time, Authors, Title, Format) |>
  kableExtra::kable(format = 'html') |>
  kableExtra::kable_classic()
```
