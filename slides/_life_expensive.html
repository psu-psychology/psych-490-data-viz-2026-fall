Visualization: Housing

p_housing <- expensive_df |>
  dplyr::filter(category == "housing") |>
  ggplot() +
  aes(x = year, y = price, color = location, 
      fill = location, shape = price_type) |>
  geom_point() + 
  geom_line(aes(x = year, y = price, 
                group = location, color = location)) +
  ylim(0, 1700000) +
  theme_classic() +
  labs(title = "Price changes in illustrative expenses: mid 1980s-present",
       subtitle = "Housing", 
       caption = "Source: Tavernise et al. (2026).") +
  scale_y_continuous(
    breaks = c(0, 500000, 1000000, 1500000),
    labels = c("0", "500,000", "1,000,000", "1,500,000")
  ) +
  ylab("price ($)") +
  theme(legend.title = element_blank()) +
  theme(text = element_text(size = 14))
