# A standard data frame with coordinates
cities_df <- data.frame(
  city = c("New York", "Paris", "Tokyo"),
  lon = c(-74.0060, 2.3522, 139.6503),
  lat = c(40.7128, 48.8566, 35.6762)
)

# Convert to an sf object (4326 is the standard GPS CRS, WGS84)
cities_sf <- st_as_sf(cities_df, coords = c("lon", "lat"), crs = 4326)

# Layer the points over the base map
ggplot() +
  geom_sf(data = world, fill = "antiquewhite", color = "gray80") +
  geom_sf(data = cities_sf, color = "red", size = 3) +
  coord_sf(xlim = c(-100, 150), ylim = c(-20, 60)) # Zooming into an area

Trim the data

pa_higher_ed_trimmed <- pa_higher_ed |>
  rename(lon = Longitude,
         lat = Latitude) |>
  select(Sector, College, Type, County, lon, lat) |>
  filter(!is.na(lat) &
         !is.na(lon))
# Convert to an sf object (4326 is the standard GPS CRS, WGS84)

pa_higher_ed_sf <-
  sf::st_as_sf(
    x = pa_higher_ed_trimmed,
    coords = c("lon", "lat"),
    crs = 4326
  )

# 3. Create or load your sample county-level dataset
# Ensure your county identifier matches the FIPS code ('GEOID' in tigris)
set.seed(123)
county_data <- data.frame(
  GEOID = county_sf$GEOID,
  value = rnorm(nrow(county_sf), mean = 50, sd = 10)
)

# 4. Join the spatial sf object with your data
map_sf <- left_join(county_sf, county_data, by = "GEOID")

# 5. Plot the choropleth map with ggplot2 and geom_sf()
ggplot(data = map_sf) +
  geom_sf(aes(fill = value), color = "white", size = 0.1) +
  scale_fill_viridis_c(option = "plasma", name = "Value") +
  theme_void() +
  labs(
    title = "County-Level Choropleth Map",
    subtitle = "Virginia Counties Colored by Sample Values",
    caption = "Source: U.S. Census Bureau / tigris"
  )
